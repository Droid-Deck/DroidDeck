cmake_minimum_required(VERSION 3.22.1)
project(droiddeck_xr_contracts LANGUAGES C CXX)

add_library(droiddeck_xr_contracts INTERFACE)
target_compile_features(droiddeck_xr_contracts INTERFACE cxx_std_17)
target_include_directories(droiddeck_xr_contracts INTERFACE
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tools/vr/src")

if(ANDROID)
    find_package(OpenXR REQUIRED CONFIG)
    add_library(droiddeck_xr_runtime STATIC OpenXrRuntime.cpp)
    set_target_properties(droiddeck_xr_runtime PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_definitions(droiddeck_xr_runtime PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES XR_USE_TIMESPEC)
    target_compile_options(droiddeck_xr_runtime PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Werror=return-type)
    target_link_libraries(droiddeck_xr_runtime PUBLIC droiddeck_xr_contracts PRIVATE OpenXR::openxr_loader EGL GLESv3 android log)
    target_include_directories(droiddeck_xr_runtime PRIVATE "${ANDROID_NDK}/sources/android/native_app_glue")
    add_library(droiddeck_xr_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
    set_target_properties(droiddeck_xr_glue PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(droiddeck_xr_glue PUBLIC "${ANDROID_NDK}/sources/android/native_app_glue")
    function(droiddeck_add_xr_activity target entry)
        add_library(${target} SHARED "${entry}")
        target_link_libraries(${target} PRIVATE droiddeck_xr_glue droiddeck_xr_runtime)
        target_link_options(${target} PRIVATE -Wl,-z,max-page-size=16384 -uANativeActivity_onCreate)
    endfunction()
endif()
