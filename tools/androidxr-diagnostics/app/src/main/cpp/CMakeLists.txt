cmake_minimum_required(VERSION 3.22.1)
project(xrdiagnostics LANGUAGES CXX)
add_library(xrdiagnostics SHARED bridge.cpp)
target_compile_features(xrdiagnostics PRIVATE cxx_std_17)
target_include_directories(xrdiagnostics PRIVATE ${CMAKE_CURRENT_LIST_DIR}/../../../../../..)
target_compile_options(xrdiagnostics PRIVATE -Wall -Wextra -Werror)
target_link_options(xrdiagnostics PRIVATE -Wl,-z,max-page-size=16384)
if(DROIDDECK_XR_FIXTURES)
    target_compile_definitions(xrdiagnostics PRIVATE DROIDDECK_XR_FIXTURES)
endif()
